### 9.1 `diff-r24.py`(§3・§4)

```python
#!/usr/bin/env python3
# 既存の 2 枚の PNG の画素差分を、領域ごとに数える。新しい起動・測定はしない。
# dec は tools/verify-kit/tools/pngdiff.py の関数をそのまま取り出して使う(import はしない。CLI の本体が走るため)。
import ast, sys, zlib, struct
import numpy as np

ROOT = '/home/ubuntu/work/signature-studio/'
src = open(ROOT + 'tools/verify-kit/tools/pngdiff.py').read()
ns = {'np': np, 'zlib': zlib, 'struct': struct}
for node in ast.parse(src).body:
    if isinstance(node, ast.FunctionDef) and node.name == 'dec':
        exec(compile(ast.Module(body=[node], type_ignores=[]), 'pngdiff-dec', 'exec'), ns)
dec = ns['dec']

# (名前, x0, x1, y0, y1)。優先順位は上から。重なる画素は先の領域に入れる。
REG = [
    ('Tempo箱 x435-492 y8-34', 435, 492, 8, 34),
    ('上部ツールバー(Tempo箱を除く) x0-1919 y0-81', 0, 1919, 0, 81),
    ('Browser窓 x0-369 y82-1079', 0, 369, 82, 1079),
    ('Channel rack x375-869 y110-204', 375, 869, 110, 204),
    ('Mixer x395-869 y548-1044', 395, 869, 548, 1044),
    ('Playlist x1040-1899 y130-1074', 1040, 1899, 130, 1074),
    ('Pattern一覧 x875-1034 y130-1059', 875, 1034, 130, 1059),
]

PAIRS = [
    ('(i) R2-4c-1 開始画 vs R2-4c-2 work-r24-run1 開始画',
     'evidence/R2-4c1/o20-start-work-r24.png', 'evidence/R2-4c2/work-r24-run1/o20-start.png'),
    ('(ii) R2-4c-2 base-run1 開始画 vs work-r24-run1 開始画',
     'evidence/R2-4c2/base-run1/o20-start.png', 'evidence/R2-4c2/work-r24-run1/o20-start.png'),
    ('(iii) R2-4c-1 開始画 vs R2-4c-2 base-run1 開始画',
     'evidence/R2-4c1/o20-start-work-r24.png', 'evidence/R2-4c2/base-run1/o20-start.png'),
]

for title, pa, pb in PAIRS:
    A = dec(ROOT + pa)
    B = dec(ROOT + pb)
    print('==', title)
    print('A:', pa, A.shape[1], 'x', A.shape[0])
    print('B:', pb, B.shape[1], 'x', B.shape[0])
    if A.shape != B.shape:
        print('different sizes', A.shape, B.shape)
        continue
    m = (np.abs(A - B).max(axis=2) > 0)
    h, w = m.shape
    n = int(m.sum())
    print('differing pixels: %d (%.4f %%)' % (n, 100.0 * n / (w * h)))
    if n == 0:
        print('no difference')
        continue
    ys, xs = np.where(m)
    print('bounding box x %d..%d, y %d..%d' % (xs.min(), xs.max(), ys.min(), ys.max()))
    lab = np.zeros((h, w), np.int32)
    for idx in range(len(REG) - 1, -1, -1):
        _, x0, x1, y0, y1 = REG[idx]
        lab[y0:y1 + 1, x0:x1 + 1] = idx + 1
    for idx, reg in enumerate(REG):
        c = int((m & (lab == idx + 1)).sum())
        print('  region %s: %d px' % (reg[0], c))
    c = int((m & (lab == 0)).sum())
    print('  region それ以外: %d px' % c)
    rows = m.sum(axis=1)
    cols = m.sum(axis=0)
    rtop = sorted([(int(rows[y]), int(y)) for y in np.nonzero(rows)[0]], key=lambda t: -t[0])[:10]
    ctop = sorted([(int(cols[x]), int(x)) for x in np.nonzero(cols)[0]], key=lambda t: -t[0])[:10]
    print('  rows top10 (y, count): ' + ', '.join('y%d:%d' % (y, c) for c, y in rtop))
    print('  cols top10 (x, count): ' + ', '.join('x%d:%d' % (x, c) for c, x in ctop))
    print('  nonzero rows: %d, nonzero cols: %d' % (int(np.count_nonzero(rows)), int(np.count_nonzero(cols))))
```

```python
#!/usr/bin/env python3
# (i) の差分を 40x20 の格子に分けて数える。dec は pngdiff.py の関数を取り出して使う。
import ast, zlib, struct
import numpy as np

ROOT = '/home/ubuntu/work/signature-studio/'
src = open(ROOT + 'tools/verify-kit/tools/pngdiff.py').read()
ns = {'np': np, 'zlib': zlib, 'struct': struct}
for node in ast.parse(src).body:
    if isinstance(node, ast.FunctionDef) and node.name == 'dec':
        exec(compile(ast.Module(body=[node], type_ignores=[]), 'pngdiff-dec', 'exec'), ns)
dec = ns['dec']

A = dec(ROOT + 'evidence/R2-4c1/o20-start-work-r24.png')
B = dec(ROOT + 'evidence/R2-4c2/work-r24-run1/o20-start.png')
m = (np.abs(A - B).max(axis=2) > 0)
ys, xs = np.where(m)
print('total', len(xs))
cw, ch = 40, 20
cells = {}
for y, x in zip(ys, xs):
    k = (int(y) // ch, int(x) // cw)
    cells[k] = cells.get(k, 0) + 1
for (cy, cx), c in sorted(cells.items(), key=lambda kv: (kv[0][1], kv[0][0])):
    print('x %4d-%4d y %3d-%3d: %4d px' % (cx * cw, cx * cw + cw - 1, cy * ch, cy * ch + ch - 1, c))
# 差分の最大値(チャンネルの差)
d = np.abs(A - B).max(axis=2)
print('max channel diff: %d, median of differing: %s' % (int(d.max()), str(np.median(d[m]))))
# 差の少なくとも 1 つで、A と B の画素値の例(上位 5 件の座標)
order = np.argsort(-d[ys, xs])[:5]
for i in order:
    y, x = int(ys[i]), int(xs[i])
    print('pix x%d y%d A=%s B=%s' % (x, y, tuple(A[y, x]), tuple(B[y, x])))
```
