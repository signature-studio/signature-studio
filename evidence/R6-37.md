### 8.1 閉包の計算(コマンドの全文)

閉包: deploy.yml と fe-build が読む入力を、args・cmd・rsp・sed・txt・tsv の中の参照で推移的に辿った、追跡ファイルの集合。起点は fe-build の文字列と deploy.yml・agent.yml の path(ファイルだけ)。非コードのファイルの中の `lift/...` の追跡ファイルを次に辿る。ディレクトリは、データが `--lists` で名指す lift/harness/hyb/lists だけ読む。

```bash
#!/usr/bin/env bash
# deploy input closure (decision 0415 §10): the tracked files that deploy.yml and fe-build read, directly or through the files they name, transitively.
# Start: the literal paths of lift/crates/fe-build/src/main.rs and the paths of .github/workflows/deploy.yml and agent.yml (files only).
# Expansion: a file that is not code (.c .h .inc .cpp .rs .py .sh .js .mjs .cjs .toml .md) contributes every `lift/...` token that is a tracked file.
# A directory is read whole only when a data file names it with --lists (lift/harness/hyb/lists).
set -u
ROOT=$(git rev-parse --show-toplevel) || exit 1
cd "$ROOT" || exit 1
W=${TMPDIR:-/tmp}/closure-work
mkdir -p "$W"
git ls-files > "$W/all.txt"
{
  grep -o 'HARNESS_DIR}/[^"]*' lift/crates/fe-build/src/main.rs | sed -E 's#^HARNESS_DIR\}/#lift/harness/#'
  grep -o '"lift/[^"]*"' lift/crates/fe-build/src/main.rs | tr -d '"'
  grep -o -E 'lift/[A-Za-z0-9_./+-]+|ci/[A-Za-z0-9_./+-]+' .github/workflows/deploy.yml .github/workflows/agent.yml
  printf '%s\n' lift/harness/two/plain-units.txt lift/harness/two/ai-units.txt
} | sed -E 's#^.*(lift/)#\1#' | sed -E 's#[^A-Za-z0-9_./+-]+$##' | sort -u > "$W/start.txt"
while read -r p; do grep -qx -F -- "$p" "$W/all.txt" && echo "$p"; done < "$W/start.txt" | sort -u > "$W/set.txt"
echo lift/harness/hyb/lists | while read -r d; do grep -E "^${d}/" "$W/all.txt"; done >> "$W/set.txt"
sort -u "$W/set.txt" -o "$W/set.txt"
while true; do
  new=$(mktemp "$W/new.XXXXXX")
  while read -r f; do
    case "$f" in
      *.c|*.h|*.inc|*.cpp|*.rs|*.py|*.sh|*.js|*.mjs|*.cjs|*.toml|*.md) continue ;;
    esac
    grep -o -E 'lift/[A-Za-z0-9_./+-]+' "$f" 2>/dev/null | sed -E 's#[^A-Za-z0-9_./+-]+$##' | sort -u | while read -r t; do
      grep -qx -F -- "$t" "$W/all.txt" && echo "$t"
    done >> "$new"
  done < <(sort -u "$W/set.txt")
  before=$(wc -l < "$W/set.txt")
  sort -u "$W/set.txt" "$new" -o "$W/set.txt"
  rm -f "$new"
  after=$(wc -l < "$W/set.txt")
  [ "$before" = "$after" ] && break
done
echo "closure: $(wc -l < "$W/set.txt") files"
echo "code: $(grep -c -E '\.(c|h|inc|cpp|rs|py|sh|js|mjs|cjs|toml)$' "$W/set.txt")"
echo "data (not code): $(grep -v -c -E '\.(c|h|inc|cpp|rs|py|sh|js|mjs|cjs|toml)$' "$W/set.txt")"
echo "python in the closure: $(grep -c '\.py$' "$W/set.txt")"
```
